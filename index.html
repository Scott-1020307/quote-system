// ⭐ 完美繼承原檔美工與排版（保留完整報價單質感） async function exportQuoteToExcel() { if (currentItems.length === 0) { alert("請先加入報價項目！"); return; } const clientName = document.getElementById('clientName').value || ''; const projectLocation = document.getElementById('projectLocation').value || ''; const contractNo = document.getElementById('contractNo').value || ''; const quoteDate = document.getElementById('quoteDate').value || ''; try { const response = await fetch('./template.xlsx'); if(!response.ok) { alert("🚨 找不到原檔範本 template.xlsx，請確認檔案已上傳至 GitHub！"); return; } const arrayBuffer = await response.arrayBuffer(); const wb = new ExcelJS.Workbook(); await wb.xlsx.load(arrayBuffer); let wsQuote = wb.getWorksheet('估價單0906') || wb.worksheets[0]; // 🛡️ 確保原檔的 FitToWidth (1頁寬) 與手動分頁被完整保留 wsQuote.pageSetup.fitToPage = true; wsQuote.pageSetup.fitToWidth = 1; wsQuote.pageSetup.fitToHeight = 0; // 高度順延 // 1. 抬頭資訊寫入 wsQuote.getCell('A6').value = `客戶名稱 : ${clientName}`; wsQuote.getCell('G6').value = contractNo; wsQuote.getCell('A7').value = `工程地點 : ${projectLocation}`; if (quoteDate) wsQuote.getCell('G7').value = quoteDate; // 2. 清空 48~95 列細項舊內容 (維持原本漂亮格式) for (let r = 48; r <= 95; r++) { let row = wsQuote.getRow(r); row.hidden = false; for (let c = 1; c <= 7; c++) { row.getCell(c).value = null; } } // 依選取項目進行工種分組 const grouped = {}; currentItems.forEach(item => { if (!grouped[item.category]) grouped[item.category] = []; grouped[item.category].push(item); }); // 5 大工種對照 const catConfig = [ { cat: "水電工程", mainRow: 11, headerRow: 48, start: 49, maxCount: 8, subtotalRow: 57 }, { cat: "燈具工程", mainRow: 12, headerRow: 59, start: 60, maxCount: 2, subtotalRow: 62 }, { cat: "木作工程", mainRow: 13, headerRow: 64, start: 65, maxCount: 7, subtotalRow: 72 }, { cat: "油漆工程", mainRow: 14, headerRow: 74, start: 75, maxCount: 8, subtotalRow: 83 }, { cat: "雜項工程", mainRow: 15, headerRow: 85, start: 86, maxCount: 6, subtotalRow: 92 } ]; const numMap = ["一", "二", "三", "四", "五", "六", "七", "八"]; const categoryTotals = { "水電工程": 0, "燈具工程": 0, "木作工程": 0, "油漆工程": 0, "雜項工程": 0 }; let activeOrderIndex = 0; wsQuote.getCell('A17').value = null; wsQuote.getCell('A19').value = null; // 3. 填入細項數據 catConfig.forEach(cfg => { let items = grouped[cfg.cat] || []; let catSum = 0; if (items.length > 0) { let numStr = numMap[activeOrderIndex] || (activeOrderIndex + 1); activeOrderIndex++; wsQuote.getCell(`A${cfg.mainRow}`).value = numStr; wsQuote.getCell(`B${cfg.mainRow}`).value = cfg.cat; wsQuote.getCell(`A${cfg.headerRow}`).value = numStr; wsQuote.getCell(`B${cfg.headerRow}`).value = cfg.cat; items.forEach((item, idx) => { if (idx < cfg.maxCount) { let rNum = cfg.start + idx; let row = wsQuote.getRow(rNum); row.getCell(2).value = item.name; row.getCell(3).value = Number(item.qty); row.getCell(4).value = item.unit; row.getCell(5).value = Number(item.price); row.getCell(6).value = Number(item.priceSubtotal); row.getCell(7).value = null; catSum += Number(item.priceSubtotal); } }); categoryTotals[cfg.cat] = catSum; wsQuote.getCell(`F${cfg.subtotalRow}`).value = catSum; } else { wsQuote.getCell(`A${cfg.mainRow}`).value = null; wsQuote.getCell(`B${cfg.mainRow}`).value = `${cfg.cat} (無施作)`; wsQuote.getCell(`A${cfg.headerRow}`).value = null; wsQuote.getCell(`B${cfg.headerRow}`).value = `${cfg.cat} (無施作)`; categoryTotals[cfg.cat] = 0; } }); // 4. 第一頁總金額 wsQuote.getCell('E11').value = categoryTotals["水電工程"]; wsQuote.getCell('F11').value = categoryTotals["水電工程"]; wsQuote.getCell('E12').value = categoryTotals["燈具工程"]; wsQuote.getCell('F12').value = categoryTotals["燈具工程"]; wsQuote.getCell('E13').value = categoryTotals["木作工程"]; wsQuote.getCell('F13').value = categoryTotals["木作工程"]; wsQuote.getCell('E14').value = categoryTotals["油漆工程"]; wsQuote.getCell('F14').value = categoryTotals["油漆工程"]; wsQuote.getCell('E15').value = categoryTotals["雜項工程"]; wsQuote.getCell('F15').value = categoryTotals["雜項工程"]; // 5. 總計與付款期別 let subtotal = currentItems.reduce((acc, i) => acc + Number(i.priceSubtotal), 0); let hasMgmt = document.getElementById('chkManagementFee').checked; let hasTax = document.getElementById('chkTax').checked; let mgmtFee = hasMgmt ? Math.round(subtotal * 0.1) : 0; let subWithMgmt = subtotal + mgmtFee; let taxFee = hasTax ? Math.round(subWithMgmt * 0.05) : 0; let grandTotal = subWithMgmt + taxFee; wsQuote.getCell('E17').value = mgmtFee; wsQuote.getCell('F18').value = subWithMgmt; wsQuote.getCell('E19').value = taxFee; wsQuote.getCell('F20').value = grandTotal; let p1 = Math.round(grandTotal * 0.3); let p2 = Math.round(grandTotal * 0.3); let p3 = Math.round(grandTotal * 0.3); let p4 = grandTotal - p1 - p2 - p3; wsQuote.getCell('F22').value = p1; wsQuote.getCell('F23').value = p2; wsQuote.getCell('F24').value = p3; wsQuote.getCell('F25').value = p4; wsQuote.getCell('C38').value = grandTotal; for (let r = 11; r <= 15; r++) { wsQuote.getRow(r).getCell(7).value = null; } // 下載產出的報價單 const buffer = await wb.xlsx.writeBuffer(); const fileName = clientName ? `裝修報價單-${clientName}-${quoteDate}.xlsx` : `裝修报價單-${quoteDate}.xlsx`; saveAs(new Blob([buffer]), fileName); } catch(err) { alert("⚠️ 套印匯出發生錯誤：" + err.message); } }
